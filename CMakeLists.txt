cmake_minimum_required(VERSION 3.21)
project(CopilotReel VERSION 0.3.6.1 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Copilot Reel requires Windows 11 and a Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# ARM64 unit tests are compiled with the app. This x64 host cannot execute them.
set(REEL_EXECUTE_TESTS ON)
if(CMAKE_VS_PLATFORM_NAME STREQUAL "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|aarch64|AARCH64")
  set(REEL_EXECUTE_TESTS OFF)
  message(STATUS "ARM64 target: unit tests are compiled and not executed on this host.")
endif()
function(reel_test name target)
  if(REEL_EXECUTE_TESTS)
    add_test(NAME ${name} COMMAND ${target})
  endif()
endfunction()
# src/vi_tokens.h is committed and regenerated from design/vi/tokens.json. A stamp is the
# output so that a clean never deletes the header; the generator rewrites it only on change.
add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/vi_tokens.stamp"
  COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen-vi-tokens.ps1"
    -TokensPath "${CMAKE_CURRENT_SOURCE_DIR}/design/vi/tokens.json"
    -OutputPath "${CMAKE_CURRENT_SOURCE_DIR}/src/vi_tokens.h"
  COMMAND ${CMAKE_COMMAND} -E touch "${CMAKE_CURRENT_BINARY_DIR}/vi_tokens.stamp"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/design/vi/tokens.json" "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen-vi-tokens.ps1"
  COMMENT "Generating src/vi_tokens.h from design/vi/tokens.json"
  VERBATIM)
add_custom_target(vi_tokens DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/vi_tokens.stamp")
add_executable(CopilotReel WIN32 src/main.cpp src/appearance.cpp src/renderer.cpp src/settings.cpp src/localization.cpp src/iconography.cpp src/osd.cpp src/behavior.cpp src/presets.cpp src/official_icons.cpp src/local_launch.cpp src/installed_apps.cpp src/artwork.cpp src/acrylic.cpp src/test_input.cpp src/tray_menu.cpp src/vi_tokens.h packaging/app.rc)
add_dependencies(CopilotReel vi_tokens)
add_custom_command(TARGET CopilotReel POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/brands"
    "$<TARGET_FILE_DIR:CopilotReel>/assets/brands"
  COMMENT "Copying official brand artwork and provenance")
target_include_directories(CopilotReel PRIVATE src packaging)
target_compile_definitions(CopilotReel PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(CopilotReel PRIVATE /utf-8 /EHsc /W4)
endif()
target_link_libraries(CopilotReel PRIVATE d3d11 d2d1 dwrite dxgi dxguid dwmapi windowsapp CoreMessaging user32 gdi32 shell32 ole32 oleaut32 propsys shlwapi comdlg32 comctl32 advapi32 wtsapi32 windowscodecs uuid imm32 uxtheme shcore)
